ملف البناء التلقائي عبر GitHub Actions

المسار: .github/workflows/build-apk.yml

name: Build Debug APK

on:
  push:
    branches:
      - main
  workflow_dispatch:

jobs:
  build:
    name: Build Android Debug APK
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Confirm AGP version 9.1.1
        run: |
          echo "=== Checking AGP version in gradle/libs.versions.toml ==="
          if [ -f "gradle/libs.versions.toml" ]; then
            if grep -E 'agp\s*=\s*"9\.1\.1"' gradle/libs.versions.toml; then
              echo "AGP version 9.1.1 confirmed in gradle/libs.versions.toml."
            else
              DETECTED=$(grep -E 'agp\s*=' gradle/libs.versions.toml || echo "not found")
              echo "Detected AGP: $DETECTED"
            fi
          else
            echo "gradle/libs.versions.toml checked."
          fi

      - name: Set up Temurin Java 17
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '17'

      - name: Set up Gradle 9.3.1
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '9.3.1'

      - name: Verify Java environment
        run: |
          echo "=== Java Version ==="
          java -version
          echo "=== JAVA_HOME ==="
          echo "JAVA_HOME: $JAVA_HOME"
          
          # Extract active Java major version
          JAVA_MAJOR=$(java -version 2>&1 | head -n 1 | awk -F '"' '{print $2}' | awk -F '.' '{print $1}')
          echo "Active Java Major Version: $JAVA_MAJOR"
          
          if [ -z "$JAVA_MAJOR" ] || [ "$JAVA_MAJOR" -lt 17 ]; then
            echo "ERROR: Java version must be 17 or higher. Detected version: $JAVA_MAJOR"
            exit 1
          fi
          echo "Java environment verification passed."

      - name: Create temporary debug keystore
        run: |
          if [ -f "./debug.keystore" ]; then
            echo "./debug.keystore already exists. Skipping creation."
          else
            echo "Generating temporary debug keystore..."
            keytool -genkeypair -v \
              -keystore ./debug.keystore \
              -storepass android \
              -alias androiddebugkey \
              -keypass android \
              -keyalg RSA \
              -keysize 2048 \
              -validity 10000 \
              -dname "CN=Android Debug,O=Android,C=US"
          fi
          
          if [ ! -f "./debug.keystore" ]; then
            echo "ERROR: Failed to verify ./debug.keystore existence."
            exit 1
          fi
          echo "Temporary debug keystore verified."

      - name: Build debug APK
        run: gradle :app:assembleDebug --stacktrace --no-daemon

      - name: Verify debug APK output
        run: |
          TARGET_APK="app/build/outputs/apk/debug/app-debug.apk"
          if [ -f "$TARGET_APK" ]; then
            echo "Found expected APK: $TARGET_APK"
            ls -lh "$TARGET_APK"
          else
            echo "ERROR: Expected APK not found at $TARGET_APK"
            echo "Searching only inside app/build/outputs/apk for any generated APKs:"
            if [ -d "app/build/outputs/apk" ]; then
              find app/build/outputs/apk -name "*.apk" -print
            else
              echo "Directory app/build/outputs/apk does not exist."
            fi
            exit 1
          fi

      - name: Upload debug APK artifact
        uses: actions/upload-artifact@v4
        with:
          name: app-debug-apk
          path: app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error

دليل الأوامر والأكواد السرية للشبكة

أوامر ADB لتفعيل الصلاحيات وأكواد Dialpad لأجهزة شاومي وسامسونج.

أمر منح الصلاحية عبر الكمبيوتر (ADB)

نفذ هذا الأمر مرة واحدة في موجه الأوامر:

adb shell pm grant com.volte.vowifi.switcher android.permission.WRITE_SECURE_SETTINGS

أمر التحقق من منح الصلاحية

adb shell dumpsys package com.volte.vowifi.switcher | grep WRITE_SECURE_SETTINGS
